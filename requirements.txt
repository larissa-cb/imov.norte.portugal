    
    # Carregar dados originais
    df = pd.read_csv('real_estate_portugal_norte.csv')
    
    # Insights por distrito
    print("\n🏆 TOP DISTRITOS PARA INVESTIMENTO:")
    district_profit = df.groupby('District').agg({
        'Price': 'mean',
        'UsableArea': 'mean'
    }).round(2)
    
    print(district_profit.sort_values('Price', ascending=False).head())
    
    # Insights por certificação energética
    print("\n🔋 IMPACTO DA CERTIFICAÇÃO ENERGÉTICA:")
    energy_impact = df.groupby('EnergyCertificate')['Price'].mean() / df['Price'].mean()
    print(energy_impact.sort_values(ascending=False))
    
    # Recomendações
    print("\n💡 RECOMENDAÇÕES PARA A AGÊNCIA:")
    print("""

    2. **Invista em certificação A/A+**: Valoriza imóvel em 15-20%

    4. **Evite**: Imóveis muito antigos (>40 anos) sem renovação

    """)
